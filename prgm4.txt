#include <iostream>
using namespace std;

class TicTacToe {
private:
    char board[3][3];
    char currentPlayer;

public:
    // Constructor
    TicTacToe() {
        currentPlayer = 'X';

        // Initialize 3x3 board
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                board[i][j] = ' ';
            }
        }
    }

    // Display game board
    void displayBoard() {
        cout << "\n";
        cout << "       TIC TAC TOE\n\n";

        for (int i = 0; i < 3; i++) {
            cout << "       ";

            for (int j = 0; j < 3; j++) {
                cout << " " << board[i][j] << " ";

                if (j < 2)
                    cout << "|";
            }

            cout << endl;

            if (i < 2)
                cout << "      ---+---+---\n";
        }

        cout << endl;
    }

    // Check whether move is valid
    bool isValidMove(int row, int col) {
        if (row < 0 || row >= 3 ||
            col < 0 || col >= 3) {
            return false;
        }

        if (board[row][col] != ' ') {
            return false;
        }

        return true;
    }

    // Place player's symbol
    void makeMove(int row, int col) {
        board[row][col] = currentPlayer;
    }

    // Check row and column wins
    bool checkRowColumn() {
        for (int i = 0; i < 3; i++) {

            // Check rows
            if (board[i][0] == currentPlayer &&
                board[i][1] == currentPlayer &&
                board[i][2] == currentPlayer) {
                return true;
            }

            // Check columns
            if (board[0][i] == currentPlayer &&
                board[1][i] == currentPlayer &&
                board[2][i] == currentPlayer) {
                return true;
            }
        }

        return false;
    }

    // Check diagonal wins
    bool checkDiagonal() {

        // Main diagonal
        if (board[0][0] == currentPlayer &&
            board[1][1] == currentPlayer &&
            board[2][2] == currentPlayer) {
            return true;
        }

        // Opposite diagonal
        if (board[0][2] == currentPlayer &&
            board[1][1] == currentPlayer &&
            board[2][0] == currentPlayer) {
            return true;
        }

        return false;
    }

    // Check winner
    bool checkWinner() {
        return checkRowColumn() || checkDiagonal();
    }

    // Check draw condition
    bool checkDraw() {
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                if (board[i][j] == ' ') {
                    return false;
                }
            }
        }

        return true;
    }

    // Switch player turns
    void switchPlayer() {
        if (currentPlayer == 'X')
            currentPlayer = 'O';
        else
            currentPlayer = 'X';
    }

    // Start game
    void playGame() {
        int row, col;

        cout << "===== TIC TAC TOE GAME =====\n";
        cout << "Player 1: X\n";
        cout << "Player 2: O\n";

        while (true) {

            displayBoard();

            cout << "Player " << currentPlayer << "'s turn\n";

            cout << "Enter row (1-3): ";
            if (!(cin >> row)) {
                cout << "Invalid input! Enter numbers only.\n";
                cin.clear();
                cin.ignore(10000, '\n');
                continue;
            }

            cout << "Enter column (1-3): ";
            if (!(cin >> col)) {
                cout << "Invalid input! Enter numbers only.\n";
                cin.clear();
                cin.ignore(10000, '\n');
                continue;
            }

            // Convert to array index
            row--;
            col--;

            // Validate move
            if (!isValidMove(row, col)) {
                cout << "\nInvalid move! Try again.\n";
                continue;
            }

            // Make move
            makeMove(row, col);

            // Check winner
            if (checkWinner()) {
                displayBoard();

                cout << "Congratulations!\n";
                cout << "Player " << currentPlayer
                     << " Wins!\n";
                break;
            }

            // Check draw
            if (checkDraw()) {
                displayBoard();
                cout << "Game Draw!\n";
                break;

            // Change player
            switchPlayer();
        }
    }
};

// Main function
int main() {
    TicTacToe game;

    game.playGame();

    return 0;
}
